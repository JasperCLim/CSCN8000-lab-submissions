In [4]:
X = credit.drop(columns='Class')
y = credit['Class'].astype(int)
X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.30, random_state=42, stratify=y
)

fraud_model = make_pipeline(
    StandardScaler(),
    LogisticRegression(max_iter=1000, class_weight='balanced', random_state=42)
)
fraud_model.fit(X_train, y_train)
fraud_predictions = fraud_model.predict(X_test)

print(f'Accuracy: {accuracy_score(y_test, fraud_predictions):.4f}')
print(classification_report(y_test, fraud_predictions, target_names=['Legitimate', 'Fraud']))
